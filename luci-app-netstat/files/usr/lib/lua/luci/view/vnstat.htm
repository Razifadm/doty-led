<%+header%> <% local sys = require "luci.sys" local http = require "luci.http" local util = require "luci.util" local uci = require "luci.model.uci".cursor() -- Format bytes to human readable local function format_bytes(bytes) if not bytes or bytes <= 0 then return "0 B" end local units = {"B", "KB", "MB", "GB", "TB"} local i = 1 while bytes >= 1024 and i < #units do bytes = bytes / 1024 i = i + 1 end return string.format("%.2f %s", bytes, units[i]) end -- Format date from mm/dd/yy to 02 Jun 25 local function format_date(date_str) local month_names = {"Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"} local m, d, y = date_str:match("(%d%d)/(%d%d)/(%d%d)") if m and d and y then m = tonumber(m) return string.format("%02d %s %02d", tonumber(d), month_names[m], tonumber(y)) end return date_str end -- Get interfaces from vnstat local function get_interfaces() local ifaces = {} local p = io.popen("ls /etc/vnstat/ 2>/dev/null | grep -v '.conf$' | grep -v '.old$'") if p then for iface in p:lines() do iface = iface:match("^%s*(.-)%s*$") if iface ~= "" then table.insert(ifaces, iface) end end p:close() end table.sort(ifaces) return ifaces end -- Parse daily stats local function get_daily_stats(iface) local stats = {} local output = sys.exec("vnstat -i " .. iface .. " --days 2>/dev/null") or "" for date, rx_val, rx_unit, tx_val, tx_unit, total_val, total_unit in output:gmatch("(%d%d/%d%d/%d%d)%s+([%d%.]+)%s+([KMGT]iB)%s+|%s+([%d%.]+)%s+([KMGT]iB)%s+|%s+([%d%.]+)%s+([KMGT]iB)") do table.insert(stats, { date = format_date(date), rx = tonumber(rx_val) * (rx_unit == "MiB" and 1024 or rx_unit == "GiB" and 1024*1024 or 1), tx = tonumber(tx_val) * (tx_unit == "MiB" and 1024 or tx_unit == "GiB" and 1024*1024 or 1), total = tonumber(total_val) * (total_unit == "MiB" and 1024 or total_unit == "GiB" and 1024*1024 or 1) }) end return stats end -- Parse monthly stats local function get_monthly_stats(iface) local stats = {} local output = sys.exec("vnstat -i " .. iface .. " --months 2>/dev/null") or "" for month, rx_val, rx_unit, tx_val, tx_unit, total_val, total_unit in output:gmatch("(%a+%s+'%d%d)%s+([%d%.]+)%s+([KMGT]iB)%s+|%s+([%d%.]+)%s+([KMGT]iB)%s+|%s+([%d%.]+)%s+([KMGT]iB)") do table.insert(stats, { month = month, rx = tonumber(rx_val) * (rx_unit == "MiB" and 1024 or rx_unit == "GiB" and 1024*1024 or 1), tx = tonumber(tx_val) * (tx_unit == "MiB" and 1024 or tx_unit == "GiB" and 1024*1024 or 1), total = tonumber(total_val) * (total_unit == "MiB" and 1024 or total_unit == "GiB" and 1024*1024 or 1) }) end return stats end local ifaces = get_interfaces() local current_iface = http.formvalue("iface") or (ifaces[1] and ifaces[1] or "") local view_mode = http.formvalue("mode") or "daily" local stats = {} local title = "" if current_iface ~= "" then if view_mode == "daily" then stats = get_daily_stats(current_iface) title = "Daily" else stats = get_monthly_stats(current_iface) title = "Monthly" end end %>
